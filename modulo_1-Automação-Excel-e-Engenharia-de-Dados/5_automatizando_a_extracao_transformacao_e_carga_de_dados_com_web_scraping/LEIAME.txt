# Projeto 3 - Automatizando Web Scraping Para Extração, Transformação e Carga de Dados

# Abra o terminal ou prompt de comando, navegue até a pasta com os arquivos e execute o comando abaixo para criar um ambiente virtual:

conda create --name dsaprocp3 python=3.13

# Ative o ambiente:

conda activate dsaprocp3 (ou: source activate dsaprocp3)

# Instale o pip e as dependências:

conda install pip
pip install -r requirements.txt 

# Execute o comando abaixo para inicializar o servidor web:

python -m http.server 8888

# Abra outro terminal, navegue até a pasta com os arquivos, inicialize o ambiente virtual (conda activate dsaprocp3) e execute o comando abaixo:

python projeto3.py

# Use os comandos abaixo para desativar o ambiente virtual e remover o ambiente (opcional):

conda deactivate
conda remove --name dsaprocp3 --all
