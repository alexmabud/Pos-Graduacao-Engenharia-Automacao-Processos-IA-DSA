# Projeto 2 - Automação do Processo de Backup, Movimentação de Arquivos, Criptografia e Logging com Python

# Abra o terminal ou prompt de comando, navegue até a pasta com os arquivos e execute o comando abaixo para criar um ambiente virtual:

conda create --name dsaprocp2 python=3.13

# Ative o ambiente:

conda activate dsaprocp2 (ou: source activate dsaprocp2)

# Instale o pip e as dependências:

conda install pip
pip install -r requirements.txt 

# Execute:

python projeto2-v1.py
python projeto2-v2.py
python projeto2-v3.py
python projeto2-v4.1.py
python projeto2-v4.2.py
python projeto2-v5.py

# Use os comandos abaixo para desativar o ambiente virtual e remover o ambiente (opcional):

conda deactivate
conda remove --name dsaprocp2 --all
