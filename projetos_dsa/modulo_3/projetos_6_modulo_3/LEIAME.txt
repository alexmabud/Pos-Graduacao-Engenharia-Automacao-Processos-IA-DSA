# Projeto 6 - Multi-Agentes de IA, RAG, Roteamento, Guardrails, Observabilidade e Explicabilidade com LangGraph, LangSmith e Pydantic LogFire

# Abra o terminal ou prompt de comando, navegue até a pasta com os arquivos e execute o comando abaixo para criar um ambiente virtual:

conda create --name dsap6 python=3.12

# Ative o ambiente:

conda activate dsap6 (ou: source activate dsap6)

# Instale o pip e as dependências:

conda install pip
pip install -r requirements.txt 

# Execute o script:

python dsa_p6_setup_rag.py

# Use os comandos abaixo para desativar o ambiente virtual e remover o ambiente (opcional):

conda deactivate
conda remove --name dsap6 --all
