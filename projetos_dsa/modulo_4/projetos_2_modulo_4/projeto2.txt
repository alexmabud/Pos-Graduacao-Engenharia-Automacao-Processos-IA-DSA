# Nó: Processamento de Dados 3 (JavaScript)

const raw = $json.data || $json || "";

// Caso 1: Se já foi analisado JSON com data[0].url
if (Array.isArray($json.data)) {
  return {
    json: {
      audioUrl: $json.data[0]?.url || null,
      rawJson: $json.data
    }
  };
}

// Caso 2: Manipular texto SSE bruto
const match = typeof raw === "string" ? raw.match(/data:\s*(\[[^\]]+\])/s) : null;

if (match) {
  try {
    const parsed = JSON.parse(match[1]);
    return {
      json: {
        audioUrl: parsed?.[0]?.url || null,
        rawJson: parsed
      }
    };
  } catch (e) {
    return {
      json: {
        error: "DSA - Falha ao analisar SSE JSON",
        raw
      }
    };
  }
}

// Caso 3: Se nenhum dos dois funcionou
return {
  json: {
    error: "DSA - URL do áudio não encontrada.",
    raw
  }
};
